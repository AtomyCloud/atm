package main

import (
 "bytes"
 "encoding/json"
 "fmt"
 "os"
 "os/exec"
 "path/filepath"
 "strconv"
 "time"
)

func main(){
 self,_:=os.Executable();dir:=filepath.Dir(self)
 if len(os.Args)>1&&os.Args[1]=="child"{os.WriteFile(filepath.Join(dir,"child.pid"),[]byte(strconv.Itoa(os.Getpid())),0600);for{time.Sleep(time.Second)}}
 if len(os.Args)>1&&os.Args[1]=="flood"{os.Stdout.Write(bytes.Repeat([]byte("x"),2<<20));return}
 if len(os.Args)>1&&os.Args[1]=="inspect"{json.NewEncoder(os.Stdout).Encode(map[string]any{"args":os.Args[2:],"env":os.Environ()});return}
 if len(os.Args)==2&&os.Args[1]=="--version"{fmt.Println("apt 2.8.3 (amd64)");return}
 if len(os.Args)==3&&os.Args[1]=="-n"&&os.Args[2]=="-v"{return}
 if filepath.Base(self)!="sudo"{fmt.Fprintln(os.Stderr,"unexpected direct mutation");os.Exit(3)}
 if len(os.Args)<6||os.Args[1]!="-n"||os.Args[2]!="--preserve-env=DEBIAN_FRONTEND"||os.Args[3]!="--"||filepath.Base(os.Args[4])!="apt-get"{fmt.Fprintln(os.Stderr,"invalid literal sudo argv");os.Exit(4)}
 os.WriteFile(filepath.Join(dir,"parent.pid"),[]byte(strconv.Itoa(os.Getpid())),0600)
 cmd:=exec.Command(self,"child");cmd.Stdout=os.Stdout;cmd.Stderr=os.Stderr;if err:=cmd.Start();err!=nil{os.Exit(5)}
 for{time.Sleep(time.Second)}
}
