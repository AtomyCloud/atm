package main

import (
 "encoding/json"
 "fmt"
 "io"
 "os"
 "os/exec"
 "path/filepath"
 "strconv"
 "strings"
 "time"
)

func main(){
 mode:=os.Getenv("ATM_TEST_MODE")
 if file:=os.Getenv("ATM_TEST_CALLS");file!=""{f,err:=os.OpenFile(file,os.O_CREATE|os.O_WRONLY|os.O_APPEND,0600);if err!=nil{panic(err)};fmt.Fprintln(f,filepath.Base(os.Args[0]));f.Close()}
 if mode=="runtime-error"{
  if filepath.Base(os.Args[0])=="yt-dlp"{fmt.Println("2025.11.12");return}
  if filepath.Base(os.Args[0])=="deno"{fmt.Print(strings.Repeat("x",70000));mode="leave-child"}else{os.Exit(75)}
 }
 switch mode{
 case "probe":
  name:=filepath.Base(os.Args[0])
  if name=="node"{fmt.Println("v22.0.0");return}
  if name=="deno"{fmt.Println("deno 2.3.0");return}
  for _,a:=range os.Args[1:]{if a=="--version"{fmt.Println("2025.11.12");return}}
  fmt.Fprintln(os.Stderr,"[debug] Optional libraries: yt_dlp_ejs-0.3.1")
  fmt.Fprintln(os.Stderr,"[debug] JS runtimes: node-22.0.0")
  if os.Getenv("PYTHONDONTWRITEBYTECODE")!="1"{os.Exit(71)}
  os.Exit(2)
 case "overflow":fmt.Print(strings.Repeat("x",70000));return
 case "child":
  if err:=os.WriteFile(os.Getenv("ATM_TEST_PID"),[]byte(strconv.Itoa(os.Getpid())),0600);err!=nil{panic(err)}
  time.Sleep(time.Hour);return
 case "parent","leave-child":
  cmd:=exec.Command(os.Args[0]);cmd.Env=append(os.Environ(),"ATM_TEST_MODE=child")
  if err:=cmd.Start();err!=nil{panic(err)}
  if file:=os.Getenv("ATM_TEST_PARENT_PID");file!=""{if err:=os.WriteFile(file,[]byte(strconv.Itoa(os.Getpid())),0600);err!=nil{panic(err)}}
  if mode=="parent"{time.Sleep(time.Hour);return}
  for i:=0;i<500;i++{if _,err:=os.Stat(os.Getenv("ATM_TEST_PID"));err==nil{return};time.Sleep(10*time.Millisecond)}
  os.Exit(72)
 }
 if mode=="chain"&&filepath.Base(os.Args[0])=="spotatui"{
  cmd:=exec.Command("yt-dlp",os.Args[1:]...);cmd.Stdin=os.Stdin;cmd.Stdout=os.Stdout;cmd.Stderr=os.Stderr
  if err:=cmd.Run();err!=nil{if e,ok:=err.(*exec.ExitError);ok{os.Exit(e.ExitCode())};fmt.Fprintln(os.Stderr,err);os.Exit(73)};return
 }
 if mode=="recursion"{
  cmd:=exec.Command(os.Getenv("ATM_TEST_SHIM"));cmd.Stdout=os.Stdout;cmd.Stderr=os.Stderr
  if err:=cmd.Run();err!=nil{os.Exit(74)};return
 }
 input,_:=io.ReadAll(os.Stdin)
 _=json.NewEncoder(os.Stdout).Encode(map[string]any{"args":os.Args[1:],"stdin":string(input),"path":os.Getenv("PATH"),"marker":os.Getenv("ATM_TEST_MARKER"),"shim_active":os.Getenv("ATM_SPOTATUI_SHIM_ACTIVE")})
 fmt.Fprintln(os.Stderr,"synthetic stderr")
 code,_:=strconv.Atoi(os.Getenv("ATM_TEST_EXIT"));os.Exit(code)
}
